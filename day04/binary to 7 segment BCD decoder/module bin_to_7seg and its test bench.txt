module bin_to_7seg 
(
  input wire [3:0] bin,

  output reg [7:1] seg 

);

  //seg from 7 to 1 is equivalent to from g to a 
  // note: this outputs are active high!!

  always @(*) begin 
    // using k mapping 
    seg[1] =  bin[1] | bin[3] | ~(bin[2] ^ bin[0]);
    seg[2] = ~bin[2] | ~(bin[0] ^ bin[1]);
    seg[3] =  bin[2] | ~bin[1] | bin[0];
    seg[4] =  bin[3] | (~bin[2] & bin[1]) | (~bin[0] & bin[1]) | (~bin[2] & ~bin[0]) | (bin[2] & ~bin[1] & bin[0]);
    seg[5] = ~bin[0] & (~bin[2] | bin[1]);
    seg[6] =  bin[3] | ~(bin[1] | bin[0]) | ( bin[2] & (~bin[1] | ~bin[0]));
    seg[7] =  bin[3] | (~bin[0] & bin[1]) | (bin[2] ^ bin[1]);


  end



endmodule
