module top_module_gray_to_7seg
(
  
  input wire [3:0] gray,

  output wire [7:1] seg

);

  // top module to make the gray code to 7 segment using the two previous modules

  wire [3:0] w;
  
  gray_bin g1
  (

    .gray(gray),

    .bin(w)


  );


  bin_to_7seg s1
  (

    .bin(w),

    .seg(seg)


  );





endmodule
