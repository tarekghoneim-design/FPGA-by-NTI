module Gray_Bin_DF
(
  input wire b3,b2,b1,b0,
  
  output wire g3,g2,g1,g0


);


  assign g3 = b3;
  assign g2 = b3 ^ b2;
  assign g1 = b2 ^ b1;
  assign g0 = b1 ^ b0;
  



endmodule

module bin_gray_shift
(
  input [3:0] bin,
  output [3:0] gray


);

assign gray = bin ^ (bin >> 1);

endmodule
