module clock_divider 
#(
    parameter N = 12  
)(
    input wire clk_in,
    input wire rst_n,
    output reg clk_out
);
 
  reg [$clog2(N)-1:0] counter;
 
always @(negedge clk_in , negedge rst_n) begin
    if (!rst_n) 
	    begin
            counter <= 0;
            clk_out <= 1;
        end 
	else 
	    begin
	
        if (counter == (N-1))
		begin
            counter <= 0;
            clk_out <= ~clk_out;  
        end 
		
		else 
		 begin
             counter <= counter + 1;
         end
    end
end
 
endmodule

`timescale 1ns / 1ps
 
module tb_clock_divider;
 
    reg clk_in;
    reg rst_n;
    wire clk_out;
 
    // Parameter for divide value 

    parameter N = 12;
 
    // Instantiate the clock divider

    clock_divider c00 
    
    (
        .clk_in(clk_in),
        .rst_n(rst_n),
        .clk_out(clk_out)
    );
 


       initial begin
        clk_in = 0;
        forever #5 clk_in = ~clk_in;
    end
 
    // Test stimulus
    initial begin
        // Initialize reset
        rst_n = 0;
        #20;
        rst_n = 1;
 
        
        #1000;
 
        $finish;
    end
 
    // Monitor signals
    initial begin
        $monitor("Time=%0t | rst_n=%b | clk_in=%b | clk_out=%b", $time, rst_n, clk_in, clk_out);
    end
 
endmodule
